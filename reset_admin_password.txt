# ==============================================================================
# OPTION 1 : EN LOCAL SUR TON PC (Sans Docker)
# Depuis le dossier ~/Documents/my_portfolio_epitech--2026
# ==============================================================================

python3 - << 'EOF'
import sqlite3, hashlib, secrets, hmac, os
from pathlib import Path

username = 'admin'
nouveau_mdp = 'MonMotDePasseLocal123!'

secret_key_path = Path('backend/data/secret_key')
secret_key = secret_key_path.read_text(encoding='utf-8').strip() if secret_key_path.exists() else os.environ.get('SECRET_KEY', '')

salt = secrets.token_hex(16)
hashed = hashlib.pbkdf2_hmac('sha256', nouveau_mdp.encode('utf-8'), salt.encode('utf-8'), 100_000).hex()
stored_hash = f"{salt}${hashed}"

integrity_sig = None
if secret_key:
    integrity_sig = hmac.new(secret_key.encode('utf-8'), f"{username}:{stored_hash}".encode('utf-8'), hashlib.sha256).hexdigest()

db_path = 'backend/data/portfolio.db'
conn = sqlite3.connect(db_path)
cur = conn.cursor()

cur.execute("PRAGMA table_info(admin_users)")
cols = [c[1] for c in cur.fetchall()]

cur.execute("SELECT id FROM admin_users WHERE username = ?", (username,))
row = cur.fetchone()

if row:
    if "integrity_signature" in cols:
        cur.execute("UPDATE admin_users SET hashed_password = ?, integrity_signature = ? WHERE username = ?", (stored_hash, integrity_sig, username))
    else:
        cur.execute("UPDATE admin_users SET hashed_password = ? WHERE username = ?", (stored_hash, username))
    print(f"✅ Mot de passe pour '{username}' mis a jour dans {db_path} !")
else:
    if "integrity_signature" in cols:
        cur.execute("INSERT INTO admin_users (username, hashed_password, integrity_signature) VALUES (?, ?, ?)", (username, stored_hash, integrity_sig))
    else:
        cur.execute("INSERT INTO admin_users (username, hashed_password) VALUES (?, ?)", (username, stored_hash))
    print(f"✅ Compte '{username}' cree avec succes dans {db_path} !")

conn.commit()
conn.close()
EOF
